// <Country> <identifier> (<local name>): describe the structure and the check-digit rule in your own words,
// and name the public source (law, standard or registry documentation).
func init() {
	Register(Detector{
		Name:        "xx-identifier",
		Label:       "<Country> <identifier> (<local name>)",
		Country:     "XX",
		OnByDefault: true,
		Pattern:     regexp.MustCompile(`\d{10}`),
		Validate:    validIdentifier,
		Keywords:    []string{"local name", "abbreviation"},
		Concepts:    []Concept{NationalID},
		Base:        domain.Low,
		Validation:  "date and check digit are valid",
	})
}

func validIdentifier(s string) bool {
	d, ok := Digits(s)
	if !ok || len(d) != 10 {
		return false
	}
	// Check the date part with ValidDate and compute the check digit here.
	return false
}

// internal/synth/xx.go

package synth

import (
	"fmt"
	"math/rand"
)

func init() {
	// Build a valid value without calling the detector: random parts, then the check digit computed here.
	Register("xx-identifier", func(r *rand.Rand) string {
		year, month, day := Date(r)
		body := Pad(year%100, 2) + Pad(month, 2) + Pad(day, 2) + Digits(r, 3)
		return body + fmt.Sprint(WeightedSum(body, []int{2, 4, 8, 5, 10, 9, 7, 3, 6})%11%10)
	})
}
